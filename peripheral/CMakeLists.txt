if(NOT DEFINED BLUEJOULE_PLATFORM AND DEFINED ENV{BLUEJOULE_PLATFORM})
  set(BLUEJOULE_PLATFORM "$ENV{BLUEJOULE_PLATFORM}")
endif()
cmake_minimum_required(VERSION 3.20.0)
if(NOT DEFINED BLUEJOULE_PLATFORM)
  message(FATAL_ERROR "Select a platform with build.py or -DBLUEJOULE_PLATFORM=<platform-id>")
endif()
set(PLATFORM_DIR "${CMAKE_CURRENT_SOURCE_DIR}/platforms/${BLUEJOULE_PLATFORM}")
if(NOT EXISTS "${PLATFORM_DIR}/platform.json")
  message(FATAL_ERROR "Unknown BlueJoule platform")
endif()
set(CONF_FILE "${CMAKE_CURRENT_SOURCE_DIR}/prj.conf;${PLATFORM_DIR}/prj.conf")
if(EXISTS "${PLATFORM_DIR}/board.overlay")
  set(DTC_OVERLAY_FILE "${PLATFORM_DIR}/board.overlay")
else()
  set(DTC_OVERLAY_FILE "")
endif()
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(bluejoule_gatt_peripheral)
target_sources(app PRIVATE src/main.c src/bj_cache_stats.c)
include("${PLATFORM_DIR}/platform.cmake")
